En ComboBox är en sökbar version av Select. Combobox är lämplig att använda när listan med möjliga val är lång eftersom det ger användaren möjlighet att söka efter ett specifikt värde.

```tsx
import { ComboBox, ListBoxItem } from '@midas-ds/components'
```

## Kontrollerat val

Använd `value` och `onChange` för att styra det valda värdet i ComboBox.

```tsx
import React from 'react'
import { Key } from '@midas-ds/components'

const [selectedFruit, setSelectedFruit] = React.useState<Key | null>(null)

return (
  <ComboBox
    label='Välj din favoritfrukt eller grönsak'
    description='Du kan bara välja en'
    value={selectedFruit}
    onChange={setSelectedFruit}
  >
    <ListBoxItem id='Lime'>Lime</ListBoxItem>
    <ListBoxItem id='Äpple'>Äpple</ListBoxItem>
    <ListBoxItem id='Banan'>Banan</ListBoxItem>
    <ListBoxItem id='Kiwi'>Kiwi</ListBoxItem>
    <ListBoxItem id='Apelsin'>Apelsin</ListBoxItem>
  </ComboBox>
)
```

## Flerval

För närvarande stödjer inte vår ComboBox flerval. Använd [Midas CSS för react-select](/dev/react-select) för detta.

## Sektioner

Vid många val kan alternativen struktureras i sektioner genom att dela in dem enligt följande:

```tsx
import { Collection, ListBoxHeader, ListBoxItem, ListBoxSection } from '@midas-ds/components'

const options = [
  {
    name: 'Grönsaker',
    children: [
      { name: 'Morot', id: 2 },
      { name: 'Broccoli', id: 3 },
      { name: 'Spenat', id: 4 },
      { name: 'Potatis', id: 5 },
      { name: 'Tomat', id: 6 },
    ],
  },
  {
    name: 'Frukter',
    children: [
      { name: 'Äpple', id: 11 },
      { name: 'Banan', id: 12 },
      { name: 'Kiwi', id: 13 },
      { name: 'Fläder', id: 14 },
      { name: 'Nektarin', id: 15 },
    ],
  },
]

return (
  <ComboBox
    label='Välj din favoritfrukt eller grönsak'
    description='Du kan bara välja en'
    items={options}
  >
    {section => (
      <ListBoxSection id={section.name}>
        <ListBoxHeader>{section.name}</ListBoxHeader>
        <Collection items={section.children}>
          {item => {
            return <ListBoxItem id={item.id}>{item.name}</ListBoxItem>
          }}
        </Collection>
      </ListBoxSection>
    )}
  </ComboBox>
)
```

:::info
`ListBox` renderas alltid via React Arias `Virtualizer`, vilket kräver en riktig webbläsares layoutmotor. I miljöer utan detta (t.ex. `happy-dom` eller `jsdom`, vanligt vid testning) kan sektionerade listor sluta rendera alternativ efter den första sektionens rubrik. Sätt `listBoxProps={{ virtualized: false }}` för att stänga av virtualisering, av testskäl eller andra anledningar.
:::

## Asynkron laddning

Om data finns på en server kan en laddningsindikator visas med en `ListBoxLoadMoreItem`-komponent.

Använd `allowsEmptyCollection` hos `ComboBox` och rendera bara `ListBoxLoadMoreItem` vid laddning för att visa ett "inga-träffar"-meddelande om sökfrågan inte gav något resultat.

```tsx
import { useAsyncList } from 'react-stately'
import { Collection, ComboBox, ListBoxItem, ListBoxLoadMoreItem, ListBoxEmptyState } from '@midas-ds/components'

const list = useAsyncList<{ name: string }>({
  async load({ signal, cursor, filterText }) {
    if (cursor) {
      cursor = cursor.replace(/^http:\/\//i, 'https://')
    }

    const res = await fetch(cursor || `https://swapi.py4e.com/api/people/?search=${filterText}`, { signal })

    const { results, next } = await res.json()

    return {
      items: results,
      cursor: next,
    }
  },
})

return (
  <ComboBox
    label='Star Wars Character Lookup'
    placeholder='Välj eller sök karaktär'
    description='Anropar ett externt API'
    inputValue={list.filterText}
    onInputChange={list.setFilterText}
    // highlight-next-line
    allowsEmptyCollection
    listBoxProps={{
      renderEmptyState: () =>
        list.isLoading ? null : (
          <ListBoxEmptyState>Inga träffar</ListBoxEmptyState>
        )
    }}
  >
    // highlight-start
    <Collection items={list.items}>
      {item => {
        return <ListBoxItem id={item.name}>{item.name}</ListBoxItem>
      }}
    </Collection>
    {list.isLoading && <ListBoxLoadMoreItem isLoading={list.isLoading} />}
    // highlight-end
  </ComboBox>
)
```

## Skapa alternativ

Använd `onAction` på `<ListBoxItem>` för att utföra valfria handlingar när användaren klickar eller trycker enter. Detta exempel visar hur användaren kan skapa ett nytt alternativ för angivet `inputValue`.

```tsx
const ItemActionExample = () => {
  const [items, setItems] = useState([
    'Lime',
    'Äpple',
    'Banan',
    'Kiwi',
    'Apelsin',
  ])
  const [inputValue, setInputValue] = useState('')
  const selectedItem = items.includes(inputValue) ? inputValue : null

  const selectItem = (key: Key | null) => {
    if (key !== null) setInputValue(String(key))
  }

  const createItem = () => setItems(prev => [...prev, inputValue])

  return (
    <ComboBox
      label='Välj din favoritfrukt eller grönsak'
      description='Skapa en ny om den ej finns i listan'
      allowsEmptyCollection
      inputValue={inputValue}
      onInputChange={setInputValue}
      value={selectedItem}
      onChange={selectItem}
    >
      {inputValue.length > 0 && selectedItem === null && (
        <ListBoxItem onAction={createItem}>
          {`Skapa "${inputValue}"`}
        </ListBoxItem>
      )}
      {items.map(item => (
        <ListBoxItem
          key={item}
          id={item}
        >
          {item}
        </ListBoxItem>
      ))}
    </ComboBox>
  )
}
```
