Filuppladdning skapas genom att kombinera [FileTrigger](/components/file-upload/file-trigger), [DropZone](/components/file-upload/dropzone) och [FileList](/components/file-upload/file-list) inuti en `FilePicker`. Nedan beskrivs mönstret för filuppladdning med dessa komponenter. Se respektive komponent för API-dokumentation.

`FilePicker` ger mönstret en etikett, en beskrivning och ett felmeddelande. En `FileTrigger` inuti beskrivs automatiskt av dessa via `aria-describedby`, så du behöver inte koppla ihop dem själv. De valda filerna hanterar du själv i ditt eget state.

## Varianter

### Standard

Standardmönstret kombinerar FileTrigger och FileList med en etikett och beskrivning. Knappen i FileTrigger blir automatiskt en sekundärknapp.

```tsx
import { Button, FilePicker, FileList, FileListItem, FileTrigger } from '@midas-ds/components'
import { Upload } from 'lucide-react'
import React from 'react'

export const FileListExample = () => {
  const [files, setFiles] = React.useState<File[]>([])

  const handleSelect = (fileList: FileList | null) => {
    const selected = fileList ? Array.from(fileList) : []
    setFiles(prev => {
      const existingNames = new Set(prev.map(f => f.name))
      return [...prev, ...selected.filter(f => !existingNames.has(f.name))]
    })
  }

  const handleDelete = (name: string) => {
    setFiles(prev => prev.filter(f => f.name !== name))
  }

  return (
    <FilePicker
      label='Ladda upp en bild på din favoritfrukt'
      description='Du kan välja flera om du har många favoritfrukter'
    >
      <FileTrigger
        allowsMultiple
        onSelect={handleSelect}
      >
        <Button icon={Upload}>Välj fil(er)</Button>
      </FileTrigger>
      {files.length > 0 && (
        <FileList aria-label='Valda filer'>
          {files.map(file => (
            <FileListItem
              key={file.name}
              fileName={file.name}
              fileSize={`${(file.size / 1024).toFixed(1)} KB`}
              onDelete={() => handleDelete(file.name)}
            />
          ))}
        </FileList>
      )}
    </FilePicker>
  )
}
```

### Med drag & drop

Genom att kombinera DropZone med FileTrigger och FileList kan användaren dra och släppa filer i en definierad yta.

```tsx
import { Button, DropZone, FilePicker, FileList, FileListItem, FileTrigger, Text } from '@midas-ds/components'
import { Upload } from 'lucide-react'
import React from 'react'
import { DropEvent, isFileDropItem } from 'react-aria'

export const DropZoneExample = () => {
  const [files, setFiles] = React.useState<File[]>([])

  const addFiles = (newFiles: File[]) => {
    setFiles(prev => {
      const existingNames = new Set(prev.map(f => f.name))
      return [...prev, ...newFiles.filter(f => !existingNames.has(f.name))]
    })
  }

  const handleDrop = async (e: DropEvent) => {
    const fileItems = e.items.filter(isFileDropItem)
    addFiles(await Promise.all(fileItems.map(item => item.getFile())))
  }

  const handleSelect = (fileList: FileList | null) => {
    if (fileList) addFiles(Array.from(fileList))
  }

  const handleDelete = (name: string) => {
    setFiles(prev => prev.filter(f => f.name !== name))
  }

  return (
    <FilePicker
      label='Ladda upp en bild på din favoritfrukt'
      description='Du kan välja flera om du har många favoritfrukter'
    >
      <DropZone onDrop={handleDrop}>
        <Text slot='label'>Välj filer eller dra och släpp filer här</Text>
        <FileTrigger
          allowsMultiple
          onSelect={handleSelect}
        >
          <Button icon={Upload}>Välj fil(er)</Button>
        </FileTrigger>
      </DropZone>
      {files.length > 0 && (
        <FileList aria-label='Valda filer'>
          {files.map(file => (
            <FileListItem
              key={file.name}
              fileName={file.name}
              fileSize={`${(file.size / 1024).toFixed(1)} KB`}
              onDelete={() => handleDelete(file.name)}
            />
          ))}
        </FileList>
      )}
    </FilePicker>
  )
}
```

<div className='card'>
  <DropZoneExample />
</div>

### Felmeddelande

Sätt `isInvalid` och `errorMessage` på `FilePicker` för att visa ett fel, till exempel om användaren väljer fel filtyp. Felmeddelandet visas alltid under kontrollerna och läggs automatiskt till i knappens beskrivning. En `DropZone` inuti markeras också som ogiltig.

```tsx
<FilePicker
  label='Ladda upp en bild på din favoritfrukt'
  isInvalid={isInvalid}
  errorMessage='Endast .jpg- och .png-filer tillåts'
>
  <DropZone onDrop={handleDrop}>
    ...
  </DropZone>
</FilePicker>
```

:::note
Lägg felmeddelandet på `FilePicker`, inte som en egen `FieldError` inuti `DropZone`. Annars visas det två gånger.
:::
